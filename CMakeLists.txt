cmake_minimum_required(VERSION 3.16)
project(liquibook_reproduction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(liquibook INTERFACE)
target_include_directories(liquibook INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_library(liquibook_simple src/simple/simple_order.cpp)
target_link_libraries(liquibook_simple PUBLIC liquibook)
find_package(Boost REQUIRED COMPONENTS unit_test_framework)
file(GLOB UNIT_SOURCES CONFIGURE_DEPENDS test/unit/ut_*.cpp)
add_executable(liquibook_unit_test ${UNIT_SOURCES})
target_compile_definitions(liquibook_unit_test PRIVATE BOOST_TEST_DYN_LINK)
target_link_libraries(liquibook_unit_test PRIVATE liquibook_simple Boost::unit_test_framework)
add_test(NAME upstream_unit_tests COMMAND liquibook_unit_test --report_level=detailed)
add_executable(pt_order_book test/perf/pt_order_book.cpp)
target_link_libraries(pt_order_book PRIVATE liquibook_simple)
add_executable(lt_order_book test/latency/lt_order_book.cpp)
target_link_libraries(lt_order_book PRIVATE liquibook_simple)
add_executable(mt_order_entry examples/mt_order_entry/mt_order_entry_main.cpp examples/mt_order_entry/Market.cpp examples/mt_order_entry/Order.cpp examples/mt_order_entry/Util.cpp)
target_link_libraries(mt_order_entry PRIVATE liquibook_simple)
add_test(NAME upstream_order_entry COMMAND mt_order_entry ${CMAKE_CURRENT_SOURCE_DIR}/examples/mt_order_entry/order_entry.script)
set_tests_properties(upstream_order_entry PROPERTIES TIMEOUT 10)

# Independent regressions exercise public APIs without the Boost test helpers.
add_executable(regression_stop_lifecycle test/regression/stop_lifecycle.cpp)
target_link_libraries(regression_stop_lifecycle PRIVATE liquibook_simple)
add_executable(regression_trade_price test/regression/trade_price.cpp)
target_link_libraries(regression_trade_price PRIVATE liquibook_simple)
add_test(NAME regression_trade_price COMMAND regression_trade_price)
foreach(action accept cancel trigger)
  foreach(side buy sell)
    add_test(NAME regression_stop_${action}_${side}
      COMMAND regression_stop_lifecycle ${action} ${side})
  endforeach()
endforeach()

option(BUILD_FIX_GATEWAY "Build the optional QuickFIX 4.4 adapter" OFF)
option(BUILD_POSTGRES_AUDIT "Build the optional asynchronous PostgreSQL audit store" OFF)
if(BUILD_POSTGRES_AUDIT)
  set(BUILD_FIX_GATEWAY ON)
endif()
if(BUILD_FIX_GATEWAY)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(QUICKFIX REQUIRED IMPORTED_TARGET quickfix)
  find_package(Threads REQUIRED)
  add_library(exchange_fix extensions/fix/adapter.cpp extensions/fix/service.cpp)
  target_link_libraries(exchange_fix PUBLIC exchange_durable PkgConfig::QUICKFIX Threads::Threads)
endif()
if(BUILD_POSTGRES_AUDIT)
  pkg_check_modules(LIBPQ REQUIRED IMPORTED_TARGET libpq)
  add_library(exchange_audit extensions/audit/audit.cpp)
  target_link_libraries(exchange_audit PUBLIC exchange_fix PkgConfig::LIBPQ)
  add_executable(audit_tests test/audit/audit_tests.cpp)
  target_compile_definitions(audit_tests PRIVATE BOOST_TEST_DYN_LINK)
  target_link_libraries(audit_tests PRIVATE exchange_audit Boost::unit_test_framework)
  add_test(NAME postgres_audit COMMAND audit_tests --report_level=detailed)
  set_tests_properties(postgres_audit PROPERTIES TIMEOUT 600 RUN_SERIAL TRUE)
  add_executable(audit_gateway examples/audit/gateway.cpp)
  target_link_libraries(audit_gateway PRIVATE exchange_audit)
  add_executable(postgres_benchmark benchmarks/postgres_benchmark.cpp)
  target_link_libraries(postgres_benchmark PRIVATE exchange_audit)
  add_executable(audit_outage_benchmark benchmarks/audit_outage_benchmark.cpp)
  target_link_libraries(audit_outage_benchmark PRIVATE exchange_audit)
endif()
if(BUILD_FIX_GATEWAY)
  add_executable(fix_tests test/fix/fix_tests.cpp)
  target_compile_definitions(fix_tests PRIVATE BOOST_TEST_DYN_LINK)
  target_link_libraries(fix_tests PRIVATE exchange_fix Boost::unit_test_framework)
  add_test(NAME fix_gateway COMMAND fix_tests --report_level=detailed)
  set_tests_properties(fix_gateway PROPERTIES TIMEOUT 300)
endif()
if(BUILD_FIX_GATEWAY)
  add_executable(fix_gateway examples/fix/gateway.cpp)
  target_link_libraries(fix_gateway PRIVATE exchange_fix)
  add_executable(fix_benchmark benchmarks/fix_benchmark.cpp)
