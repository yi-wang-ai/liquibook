cmake_minimum_required(VERSION 3.16)
project(liquibook_reproduction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(liquibook INTERFACE)
target_include_directories(liquibook INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_library(liquibook_simple src/simple/simple_order.cpp)
target_link_libraries(liquibook_simple PUBLIC liquibook)
find_package(Boost REQUIRED COMPONENTS unit_test_framework)
file(GLOB UNIT_SOURCES CONFIGURE_DEPENDS test/unit/ut_*.cpp)
add_executable(liquibook_unit_test ${UNIT_SOURCES})
target_compile_definitions(liquibook_unit_test PRIVATE BOOST_TEST_DYN_LINK)
target_link_libraries(liquibook_unit_test PRIVATE liquibook_simple Boost::unit_test_framework)
add_test(NAME upstream_unit_tests COMMAND liquibook_unit_test --report_level=detailed)
add_executable(pt_order_book test/perf/pt_order_book.cpp)
target_link_libraries(pt_order_book PRIVATE liquibook_simple)
add_executable(lt_order_book test/latency/lt_order_book.cpp)
target_link_libraries(lt_order_book PRIVATE liquibook_simple)
add_executable(mt_order_entry examples/mt_order_entry/mt_order_entry_main.cpp examples/mt_order_entry/Market.cpp examples/mt_order_entry/Order.cpp examples/mt_order_entry/Util.cpp)
target_link_libraries(mt_order_entry PRIVATE liquibook_simple)
add_test(NAME upstream_order_entry COMMAND mt_order_entry ${CMAKE_CURRENT_SOURCE_DIR}/examples/mt_order_entry/order_entry.script)
set_tests_properties(upstream_order_entry PROPERTIES TIMEOUT 10)

# Independent regressions exercise public APIs without the Boost test helpers.
add_executable(regression_stop_lifecycle test/regression/stop_lifecycle.cpp)
target_link_libraries(regression_stop_lifecycle PRIVATE liquibook_simple)
add_executable(regression_trade_price test/regression/trade_price.cpp)
target_link_libraries(regression_trade_price PRIVATE liquibook_simple)
add_test(NAME regression_trade_price COMMAND regression_trade_price)
foreach(action accept cancel trigger)
  foreach(side buy sell)
    add_test(NAME regression_stop_${action}_${side}
      COMMAND regression_stop_lifecycle ${action} ${side})
  endforeach()
endforeach()
